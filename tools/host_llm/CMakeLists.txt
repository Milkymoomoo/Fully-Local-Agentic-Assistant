cmake_minimum_required(VERSION 3.22)
project(host_llm C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SRC ${CMAKE_CURRENT_LIST_DIR}/../../app/src/main/cpp)
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../../third_party/llama.cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(o LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_OPENMP)
  set(${o} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${LLAMA_SRC} build-llama)
add_executable(host_llm main.cpp ${SRC}/llm_core.cpp)
target_include_directories(host_llm PRIVATE ${SRC} ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include)
target_link_libraries(host_llm llama)
